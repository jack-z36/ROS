# 大象夹爪Ubuntu上位机控制行动框架

## 目标
使用Ubuntu上位机，通过USB转RS485模块，控制Elephant Robotics myCobot Pro力控夹爪（myGripper-F100）实现开合动作。

## 阶段一：Ubuntu软件环境搭建

> [!info] 核心思路
> 我们需要一个“翻译官”来帮我们和夹爪说话。这个翻译官就是Python的 `pyserial` 库，它能直接操作串口，发送十六进制指令。

### 1.1 安装Python和pip
打开Ubuntu终端（快捷键：`Ctrl`+`Alt`+`T`），执行：
```bash
sudo apt update
sudo apt install python3 python3-pip -y
```

### 1.2 安装串口通信库
```bash
pip3 install pyserial
```

## 阶段二：验证基本通信

> [!info] 核心思路
> 在正式控制开合前，先试试“对话”是否通畅。我们将使用原始串口通信，直接发送十六进制指令，参考[[自定义协议控制]]中的协议格式。

### 2.1 识别串口设备
在终端执行：
```bash
ls /dev/ttyUSB*
```
- 如果看到类似 `/dev/ttyUSB0` 的输出，说明模块已被识别。
- 如果没有，请检查驱动是否安装正确，或尝试更换USB口。

### 2.2 确认串口权限
为了能正常读写串口，可能需要将当前用户加入 `dialout` 组：
```bash
sudo usermod -a -G dialout $USER
```
执行后，**需要注销并重新登录Ubuntu**才能生效。

### 2.3 通信测试脚本
创建一个Python文件 `test_communication.py`，使用原始串口通信验证：
```python
import serial
import time
import binascii

# !!! 重要：修改为你实际的串口号 !!!
SERIAL_PORT = "/dev/ttyUSB0"
BAUDRATE = 115200
TIMEOUT = 1  # 超时时间（秒）

def calculate_crc16_modbus(data):
    """计算CRC16 Modbus校验码"""
    crc = 0xFFFF
    for byte in data:
        crc ^= byte
        for _ in range(8):
            if crc & 0x0001:
                crc = (crc >> 1) ^ 0xA001
            else:
                crc >>= 1
    return crc

def send_command(ser, command_hex):
    """发送十六进制指令并接收返回"""
    # 将十六进制字符串转换为字节数组
    command_bytes = bytes.fromhex(command_hex)
    print(f"发送: {command_hex}")
    
    # 发送指令
    ser.write(command_bytes)
    time.sleep(0.1)  # 等待响应
    
    # 读取响应
    if ser.in_waiting > 0:
        response = ser.read(ser.in_waiting)
        response_hex = binascii.hexlify(response).decode('utf-8').upper()
        print(f"接收: {response_hex}")
        return response_hex
    else:
        print("未接收到响应")
        return None

def test_gripper():
    """测试夹爪通信"""
    print(f"正在尝试连接夹爪: {SERIAL_PORT}")
    try:
        # 打开串口
        ser = serial.Serial(
            port=SERIAL_PORT,
            baudrate=BAUDRATE,
            timeout=TIMEOUT
        )
        print("✅ 串口打开成功")
        
        # 测试1: 读取固件主版本号
        # 指令: fe fe 08 0e 03 00 01 00 00 72 51
        print("\n--- 测试1: 读取固件主版本号 ---")
        response1 = send_command(ser, "FEFE080E0300010000")
        
        # 测试2: 读取设备ID号
        # 指令: FE FE 08 0E 03 00 04 00 00 73 41
        print("\n--- 测试2: 读取设备ID号 ---")
        response2 = send_command(ser, "FEFE080E0300040000")
        
        # 关闭串口
        ser.close()
        print("\n✅ 串口关闭成功")
        
        # 简单分析响应
        if response1 and response2:
            print("\n✅ 通信测试完成！")
            print("请观察返回的数据是否符合协议格式。")
            print("参考协议格式: 帧头(FE FE) + 长度(08) + ID(0E) + 功能码(03) + 寄存器地址 + 数据 + CRC校验")
        else:
            print("\n❌ 通信测试失败，未收到完整响应。")
            
    except serial.SerialException as e:
        print(f"❌ 串口打开失败: {e}")
        print("请检查：1. 串口号是否正确 2. 设备是否连接 3. 权限是否正确")
    except Exception as e:
        print(f"❌ 测试过程中出错: {e}")

if __name__ == "__main__":
    test_gripper()
```

### 2.4 运行测试
在终端执行：
```bash
python3 test_communication.py
```
**预期结果**：
- 看到“✅ 串口打开成功”
- 看到发送和接收的十六进制数据
- 接收数据应符合协议格式（以FE FE开头）

**如果失败**：根据错误提示，回头检查串口号、权限、接线等。

### 2.5 协议解析（可选）
如果你想手动解析返回数据，可以参考[[自定义协议控制]]中的协议格式：
- **帧头**：FE FE（固定）
- **长度**：08（固定）
- **ID**：0E（默认14）
- **功能码**：03（读操作）或06（写操作）
- **寄存器地址**：2字节，具体功能见寄存器表
- **数据**：2字节，返回的数据
- **CRC校验**：2字节，用于验证数据完整性

## 阶段三：控制夹爪开合

> [!info] 核心思路
> 通信成功后，我们就可以发送“开”和“合”的指令了。角度范围是0（全闭）到100（全开）。

### 3.1 控制脚本
创建一个Python文件 `control_gripper.py`：
```python
import serial
import time
import binascii

# !!! 修改为你实际的串口号和夹爪ID !!!
SERIAL_PORT = "/dev/ttyUSB0"
GRIPPER_ID = 14  # 默认ID为14，即0x0E
BAUDRATE = 115200
TIMEOUT = 1

def calculate_crc16_modbus(data):
    """计算CRC16 Modbus校验码"""
    crc = 0xFFFF
    for byte in data:
        crc ^= byte
        for _ in range(8):
            if crc & 0x0001:
                crc = (crc >> 1) & 0xA001
            else:
                crc >>= 1
    return crc

def build_command(function_code, register_address, data=0x0000):
    """构建发送指令"""
    # 帧头
    frame_header = [0xFE, 0xFE]
    # 长度（固定08）
    length = [0x08]
    # ID
    gripper_id = [GRIPPER_ID]
    # 功能码
    func_code = [function_code]
    # 寄存器地址（2字节）
    reg_addr = [(register_address >> 8) & 0xFF, register_address & 0xFF]
    # 数据（2字节）
    data_bytes = [(data >> 8) & 0xFF, data & 0xFF]
    
    # 组合数据（不含CRC）
    data_list = frame_header + length + gripper_id + func_code + reg_addr + data_bytes
    
    # 计算CRC
    crc = calculate_crc16_modbus(data_list)
    crc_bytes = [(crc >> 8) & 0xFF, crc & 0xFF]
    
    # 完整指令
    full_command = data_list + crc_bytes
    return ''.join([f'{byte:02X}' for byte in full_command])

def send_command(ser, command_hex):
    """发送十六进制指令并接收返回"""
    command_bytes = bytes.fromhex(command_hex)
    print(f"发送: {command_hex}")
    
    ser.write(command_bytes)
    time.sleep(0.1)
    
    if ser.in_waiting > 0:
        response = ser.read(ser.in_waiting)
        response_hex = binascii.hexlify(response).decode('utf-8').upper()
        print(f"接收: {response_hex}")
        return response_hex
    else:
        print("未接收到响应")
        return None

def set_gripper_enable(ser, enable):
    """设置夹爪使能状态"""
    # 寄存器地址: 0x000A (10)
    # enable: 0=掉使能, 1=上使能
    cmd = build_command(0x06, 0x000A, enable)
    return send_command(ser, cmd)

def set_gripper_angle(ser, angle, speed=50):
    """设置夹爪角度"""
    # 寄存器地址: 0x000B (11)
    # angle: 0-100
    cmd = build_command(0x06, 0x000B, angle)
    return send_command(ser, cmd)

def set_gripper_speed(ser, speed):
    """设置夹爪速度"""
    # 寄存器地址: 0x0020 (32)
    # speed: 1-100
    cmd = build_command(0x06, 0x0020, speed)
    return send_command(ser, cmd)

def main():
    try:
        # 打开串口
        ser = serial.Serial(
            port=SERIAL_PORT,
            baudrate=BAUDRATE,
            timeout=TIMEOUT
        )
        print("✅ 串口打开成功")
        
        # 1. 使能夹爪
        print("\n--- 1. 使能夹爪 ---")
        set_gripper_enable(ser, 1)  # 1=上使能
        time.sleep(0.5)
        
        # 2. 设置速度（可选）
        print("\n--- 2. 设置速度为50 ---")
        set_gripper_speed(ser, 50)
        time.sleep(0.5)
        
        # 3. 打开夹爪到100%（全开）
        print("\n--- 3. 打开夹爪到100% ---")
        set_gripper_angle(ser, 100)
        time.sleep(2)  # 等待夹爪运动到位
        
        # 4. 关闭夹爪到0%（全闭）
        print("\n--- 4. 关闭夹爪到0% ---")
        set_gripper_angle(ser, 0)
        time.sleep(2)
        
        # 5. 断使能
        print("\n--- 5. 断使能夹爪 ---")
        set_gripper_enable(ser, 0)  # 0=掉使能
        
        # 关闭串口
        ser.close()
        print("\n✅ 控制完成！夹爪应该已经完成了一次开合动作。")
        
    except serial.SerialException as e:
        print(f"❌ 串口打开失败: {e}")
    except Exception as e:
        print(f"❌ 控制过程中出错: {e}")

if __name__ == "__main__":
    main()
```

### 3.2 运行控制
```bash
python3 control_gripper.py
```
**观察**：夹爪应该先全开，等待2秒，然后全闭。

## 阶段四：参数调整与进阶

> [!tip] 常用参数调整
> 你可以通过修改脚本中的参数来调整夹爪的行为。

| 参数 | 寄存器地址 | 范围 | 说明 |
|------|------------|------|------|
| **速度** | 0x0020 (32) | 1-100 | 数值越大，运动越快 |
| **扭矩** | 0x001B (27) | 0-100 | 数值越大，夹持力越大 |
| **角度** | 0x000B (11) | 0-100 | 0=全闭，100=全开 |

## 阶段五：常见问题与调试

> [!question] 遇到问题？
> 别慌，按照以下步骤排查。

1.  **通信失败**：
    - 检查串口号：`ls /dev/ttyUSB*`
    - 检查权限：`ls -l /dev/ttyUSB0`，看是否属于 `dialout` 组
    - 检查接线：A-A，B-B，不要反接
    - 检查电源：夹爪必须用24V外部电源供电

2.  **夹爪不动**：
    - 确认已执行使能指令（寄存器0x000A写入1）
    - 检查速度设置是否为0
    - 检查扭矩设置是否过小

3.  **运动抖动**：
    - 尝试调整PID参数（D值），适当增大D值可减少抖动
    - 参考[[自定义协议控制]]中的PID设置指令
